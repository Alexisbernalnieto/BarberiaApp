rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // --- FUNCIONES DE AYUDA ---
    
    // Verifica que el documento del usuario no esté suspendido ni eliminado
    function isActiveUser() {
      let userDoc = get(/databases/$(database)/documents/users/$(request.auth.uid));
      return userDoc == null || !('status' in userDoc.data) || userDoc.data.status == 'active';
    }

    function isSignedIn() {
      return request.auth != null && isActiveUser();
    }
    
    // --- NUEVO SISTEMA CUSTOM CLAIMS (Rápido, Barato y Seguro) ---
    function hasRole(allowedRoles) {
      return ('role' in request.auth.token) && (request.auth.token.role in allowedRoles);
    }
    
    function isAdmin() {
      let uid = request.auth.uid;
      let userDoc = get(/databases/$(database)/documents/users/$(uid));
      return isSignedIn() && (
        hasRole(['admin', 0]) || 
        (userDoc != null && (userDoc.data.role == 'admin' || userDoc.data.role == 0)) ||
        (('email' in request.auth.token) && (
          request.auth.token.email.lower() == 'alexisb.ti22@utsjr.edu.mx' || 
          request.auth.token.email.lower() == 'alexisbn.ti22@utsjr.edu.mx' ||
          request.auth.token.email.lower() == 'alexisbernalnieto@gmail.com'
        ))
      );
    }
    
    function isReception() {
      let uid = request.auth.uid;
      let userDoc = get(/databases/$(database)/documents/users/$(uid));
      return isSignedIn() && (
        hasRole(['reception', 2]) ||
        (userDoc != null && (userDoc.data.role == 'reception' || userDoc.data.role == 2))
      );
    }
    
    function isBarber() {
      let uid = request.auth.uid;
      let userDoc = get(/databases/$(database)/documents/users/$(uid));
      return isSignedIn() && (
        hasRole(['barber', 3]) ||
        (userDoc != null && (userDoc.data.role == 'barber' || userDoc.data.role == 3))
      );
    }
    
    function isStaff() {
      return isAdmin() || isReception() || isBarber();
    }
    
    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    // --- REGLAS POR COLECCIÓN ---

    // USUARIOS
    match /users/{userId} {
      // Cualquiera puede registrarse (crear su propio doc)
      allow create: if (isSignedIn() && request.auth.uid == userId) || isAdmin();
      
      // Leer: 
      // - El propio usuario
      // - Admin y Recepción (para gestión)
      // - Barberos pueden leer usuarios para ver nombres de clientes
      // - Cualquier usuario puede leer perfiles de barberos (para el sistema de citas)
      allow read: if isOwner(userId) || isAdmin() || isReception() || isBarber() || (resource.data != null && resource.data.role in [3, 'barber']); 
      
      // Actualizar:
      // - Admin (puede cambiar roles)
      // - Dueño (puede editar nombre/teléfono, pero NO puede cambiar su rol, y debe estar activo)
      allow update: if isAdmin() || 
        (isOwner(userId) && isActiveUser() && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['role']));
      
      // Eliminar: Solo Admin
      allow delete: if isAdmin();
    }

    // --- VALIDACIONES DE ESQUEMA ---
    function isValidAppointment() {
      let data = request.resource.data;
      return data.keys().hasAll(['userId', 'userName', 'barberId', 'date', 'time', 'serviceId', 'price', 'status']) &&
             data.price >= 0 &&
             data.status in ['pending_payment', 'confirmed', 'Confirmado', 'completed', 'cancelled', 'checked_in', 'in_progress', 'no_show', 'unhandled', 'rescheduled', 'En Local'];
    }
    
    function isValidLog() {
       let data = request.resource.data;
       return data.keys().hasAll(['adminId', 'action', 'timestamp']) &&
              data.adminId == request.auth.uid; // El usuario solo puede registrar logs a su nombre
    }

    // CITAS (Appointments)
    match /appointments/{appointmentId} {
      // Leer: Permitir a todos los usuarios autenticados leer las citas.
      // Esto es estrictamente necesario porque el frontend necesita cargar todas las 
      // citas de un barbero en una fecha para calcular la disponibilidad.
      // Si se restringe por estado en las reglas pero no en el frontend, 
      // Firestore rechaza la consulta ("Rules are not filters").
      allow read: if isSignedIn();

      allow create: if isSignedIn() && isValidAppointment() && (
        // Regla 1: Un cliente puede crear su propia cita
        (request.resource.data.userId == request.auth.uid && (
          // Si el estatus es paid debe traer stripe intent de lo contrario pasa sin el intent
          (request.resource.data.paid == true && request.resource.data.paymentIntentId != null) ||
          (request.resource.data.paid == false && request.resource.data.paymentIntentId == null)
        )) ||
        
        // Regla 2: Staff puede crear citas (ej. Walk-ins)
        isStaff()
      );
      
      // Actualizar:
      allow update: if isSignedIn() && (
        // Solo Admin, Recepción o el propio Barbero pueden actualizar campos de seguimiento
        (isStaff() && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'paid', 'paymentMethod', 'status', 'paidAt', 'updatedAt', 
          'checkedInAt', 'inProgressAt', 'completedAt', 'noShowMarkedAt', 
          'cancelledAt', 'cancellationReason', 'cancellationType', 'unhandledAt',
          'notes'
        ])) ||
        
        // El dueño puede cancelar su propia cita
        (isOwner(resource.data.userId) && request.resource.data.status == 'cancelled') ||
        
        // El dueño puede enviar justificación de no-show
        (isOwner(resource.data.userId) && resource.data.status == 'no_show' && 
         request.resource.data.diff(resource.data).affectedKeys().hasOnly([
           'noShowJustification', 'rescheduleRequested', 'updatedAt'
         ]))
      );
      
      // Eliminar: Solo Admin o Recepción
      allow delete: if isAdmin() || isReception();
    }

    // SERVICIOS
    match /services/{serviceId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // SUCURSALES
    match /branches/{branchId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // BARBEROS PUBLICOS
    match /barbers/{barberId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // REGISTROS DIARIOS DE BARBEROS (METRICAS)
    match /barber_day_logs/{logId} {
      // Staff puede leer sus propios logs y admin todos
      allow read: if isSignedIn() && isStaff();
      
      // Crear/Actualizar: Solo el propio barbero o Admin
      allow create, update: if isSignedIn() && (
        isAdmin() || 
        (isBarber() && (
          // El ID del documento suele ser log_UID_date, o el campo barberId coincide
          request.resource.data.barberId == request.auth.uid
        ))
      );
    }

    // NOTIFICACIONES
    match /notifications/{notificationId} {
      // Staff puede leer notificaciones dirigidas a su rol
      allow read: if isSignedIn() && isStaff();
      
      // Permitir creación a cualquier usuario firmado (para notificar al staff sobre nuevas citas/cancelaciones)
      allow create: if isSignedIn();
      
      // Staff puede actualizar (marcar como leídas)
      allow update: if isSignedIn() && isStaff();
      
      // Solo Admin puede eliminar notificaciones
      allow delete: if isAdmin();
    }
    
    // HISTORIAL / LOGS
    match /activity_logs/{logId} {
      // Reglas estrictas:
      // Lectura: Únicamente los Administradores (para que los clientes no puedan espiar).
      allow read: if isAdmin();
      
      // Creación: Operaciones permitidas solo para la administración y staff de la Barbería, o clientes que validan su propio ID.
      // Validación estricta para evitar inyección de logs falsos.
      allow create: if isSignedIn() && isValidLog() && (isStaff() || request.auth.uid == request.resource.data.adminId);

      // Mantenimiento de auditoría (INMUTABLE):
      // Nadie, absolutamente nadie, tiene permitido actualizar o destruir registros del sistema de Logs.
      allow update, delete: if false;
    }
    
    // Regla por defecto: Bloquear todo lo demás
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
